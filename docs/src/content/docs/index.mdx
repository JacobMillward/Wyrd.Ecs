import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
	<Card title="Parallel by default" icon="random">
		Register systems with `WorldBuilder`. The scheduler groups what can run together and spreads it across your cores.
	</Card>
	<Card title="Live debug UI" icon="magnifier">
		`world.WithDebugServer()`. Browse archetypes and entities, edit components live, and watch a running change log from your browser.
	</Card>
	<Card title="AOT-compatible" icon="approve-check">
		Fully source-generated throughout. No reflection, no dynamic codegen.
	</Card>
	<Card title="First-class persistence" icon="cloud-download">
		Mark your components, reference a codec package, one method on `WorldBuilder`. Binary, JSON, or continuous WAL.
	</Card>
	<Card title="Archetype storage" icon="puzzle">
		Entities with the same components live together in dense arrays. No boxing, no reflection on the hot path.
	</Card>
	<Card title="Source-generated queries" icon="rocket">
		`world.Query().With<Position, Velocity>().ForEach(...)`. Access comes from `ref`/`in` on the callback itself.
	</Card>
	<Card title="Runtime control" icon="setting">
		Enable, disable, add, or remove systems at runtime. Ordering resolves correctly regardless of registration order.
	</Card>
	<Card title="Hierarchy & templates" icon="list-format">
		Parent/child hierarchy and `EntityTemplate` prefabs built in. Instantiate a whole subtree in one call.
	</Card>
</CardGrid>
